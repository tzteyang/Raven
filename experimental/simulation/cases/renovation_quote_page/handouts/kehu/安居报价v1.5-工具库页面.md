  body { margin: 0; background: var(--bg); color: var(--ink); font-size: 15px;

  tr.sum td { font-weight: 600; }
  tr.total td { padding-top: 8px; font-size: 18px; font-weight: 700; border-bottom: none; color: var(--brand); }

<script>
(function () {
  var LISTS = [{"code": "AJ-JG-2026-04", "start": "2026-10-01", "units": {"jz": 680, "bz": 980, "jzh": 1380}, "demolish": 45, "custom": 28000}];
  var PROMOS = [{"code": "AJ-YX-2026-10", "name": "金秋家装节", "start": "2026-10-01", "end": "2026-10-31", "tiers": [[100000, 3000]]}, {"code": "AJ-YX-2026-11", "name": "暖冬安家", "start": "2026-11-01", "end": "2026-11-30", "tiers": [[150000, 6000], [80000, 2000]]}];
  var NAMES = { jz: '简装', bz: '标准', jzh: '精装' };
  var MSG = '请输入 1 到 1000 之间的面积';
  function val(id) { return document.getElementById(id).value.trim(); }
  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); }
  function pad(n) { return (n < 10 ? '0' : '') + n; }
  function ymd(d) { return d.getFullYear() + '-' + pad(d.getMonth() + 1) + '-' + pad(d.getDate()); }
  function group(n) { return String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }
  function money(fen) { return group(Math.floor(fen / 100)) + '.' + String(fen % 100 + 100).slice(1); }
  function sqm(h) { return group(Math.floor(h / 100)) + '.' + String(h % 100 + 100).slice(1); }
  function toHalf(s) {
    return s.replace(/[\uff01-\uff5e]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0xfee0); })
            .replace(/\u3000/g, ' ');
  }
  function readArea(raw) {
    var s = toHalf(String(raw || '')).replace(/\s+/g, '');
    s = s.replace(/(平方米|平米|平方|㎡|m²|平)$/, '');
    if (!/^\d+(\.\d{1,2})?$/.test(s)) return null;
    var p = s.split('.');
    var h = parseInt(p[0], 10) * 100 + (p[1] ? parseInt((p[1] + '0').slice(0, 2), 10) : 0);
    if (h < 100 || h > 100000) return null;
    return h;
  }
  function compute(h, tier, demolish, custom, list, promo) {
    var billed = Math.max(h, 6000);
    var base = billed * list.units[tier];
    var demo = demolish ? billed * list.demolish : 0;
    var cust = custom ? list.custom * 100 : 0;
    var sub = base + demo + cust;
    var mgmt = Math.floor((sub * 8 + 50) / 100);
    var tax = Math.floor(((sub + mgmt) * 9 + 50) / 100);
    var gross = sub + mgmt + tax;
    var cut = 0;
    if (promo) {
      for (var i = 0; i < promo.tiers.length; i++) {
        if (gross >= promo.tiers[i][0] * 100) { cut = promo.tiers[i][1] * 100; break; }
      }
    }
    return { billed: billed, base: base, demo: demo, cust: cust, mgmt: mgmt, tax: tax, gross: gross, cut: cut,
             total: Math.floor((gross - cut + 50) / 100) };
  }
  function inForce(day) {
    var list = null, promo = null, i;
    for (i = 0; i < LISTS.length; i++) { if (LISTS[i].start <= day) list = LISTS[i]; }
    for (i = 0; i < PROMOS.length; i++) { if (PROMOS[i].start <= day && day <= PROMOS[i].end) promo = PROMOS[i]; }
    return { list: list, promo: promo };
  }
  ['adv', 'advtel'].forEach(function (id) {
    var el = document.getElementById(id);
    try { el.value = localStorage.getItem('anju-' + id) || ''; } catch (e) {}
    el.addEventListener('change', function () { try { localStorage.setItem('anju-' + id, el.value); } catch (e) {} });
  });
  function show() {
    var err = document.getElementById('err'), sheet = document.getElementById('sheet');
    var h = readArea(document.getElementById('area').value);
    if (h === null) {
      err.textContent = MSG; err.style.display = 'block'; sheet.style.display = 'none'; sheet.innerHTML = '';
      return;
    }
    err.style.display = 'none';
    var now = new Date(), day = ymd(now);
    var until = ymd(new Date(now.getFullYear(), now.getMonth(), now.getDate() + 7));
    var force = inForce(day), tier = document.getElementById('tier').value;
    var demolish = document.getElementById('demolish').checked, custom = document.getElementById('custom').checked;
    var q = compute(h, tier, demolish, custom, force.list, force.promo);
    var no = 'AJ' + day.replace(/-/g, '') + '-' + pad(now.getHours()) + pad(now.getMinutes()) + pad(now.getSeconds());
    var cut = force.promo ? (q.cut ? '-' + money(q.cut) : '未达活动门槛') : '无';
    var promoName = force.promo ? '（' + force.promo.name + '）' : '';
    sheet.innerHTML = '<h2>安居家装 整装报价单</h2>'
      + '<div class="meta"><span>编号 ' + no + '</span><span>报价日期 ' + day + '</span></div>'
      + '<div class="meta"><span>有效期至 ' + until + '</span><span>价目表 ' + force.list.code + '</span></div>'
      + '<div class="meta"><span>家装顾问 ' + esc(val('adv') || '—') + '</span><span>' + esc(val('advtel') || '') + '</span></div>'
      + '<table>'
      + '<tr><td>建筑面积</td><td class="amt">' + sqm(h) + ' ㎡</td></tr>'
      + '<tr><td>计价面积</td><td class="amt">' + sqm(q.billed) + ' ㎡' + (h < 6000 ? '（不足 60㎡ 按 60㎡ 计）' : '') + '</td></tr>'
      + '<tr><td>档次</td><td class="amt">' + NAMES[tier] + ' ' + force.list.units[tier] + ' 元/㎡</td></tr>'
      + '<tr><td>基础装修费</td><td class="amt">' + money(q.base) + '</td></tr>'
      + '<tr><td>拆旧费</td><td class="amt">' + (demolish ? money(q.demo) : '不拆旧') + '</td></tr>'
      + '<tr><td>全屋定制费</td><td class="amt">' + (custom ? money(q.cust) : '不做定制') + '</td></tr>'
      + '<tr><td>管理费（8%）</td><td class="amt">' + money(q.mgmt) + '</td></tr>'
      + '<tr><td>税金（9%）</td><td class="amt">' + money(q.tax) + '</td></tr>'
      + '<tr class="sum"><td>含税合计</td><td class="amt">' + money(q.gross) + '</td></tr>'
      + '<tr><td>活动立减' + promoName + '</td><td class="amt">' + cut + '</td></tr>'
      + '<tr class="total"><td>报价合计</td><td class="amt">' + group(q.total) + ' 元</td></tr>'
      + '</table>'
      + '<div class="note">以量房实测面积为准。本报价自报价日起 7 天内有效。<br>'
      + '价目表 ' + force.list.code + (force.promo ? ' · 活动 ' + force.promo.code : ' · 当日无报价立减活动') + '</div>';
    sheet.style.display = 'block';
  }
  document.getElementById('go').addEventListener('click', show);
})();
</script>
</body>
</html>
